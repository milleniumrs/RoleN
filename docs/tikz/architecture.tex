\begin{tikzpicture}[

  prov/.style       = {box, fill=white, minimum width=4.6cm, minimum height=1.1cm},

  lanelbl/.style    = {font=\scriptsize\bfseries, text=black!55, anchor=west},
  flow/.style       = {-{Latex[length=4pt]}, draw=black!55, line width=0.6pt},
  ticket/.style     = {-{Latex[length=5pt]}, draw=black!80, line width=1.4pt},
  side/.style       = {-{Latex[length=3pt]}, draw=black!35, line width=0.4pt, dashed},
  cap/.style        = {font=\scriptsize\itshape, text=black!60},
]

% ---------------- front-ends ----------------
\node[front] (tui) at (3.4,10.4) {\textbf{maestro-tui} (AppCUI)\\[-1pt]{\scriptsize dashboard · providers · rules · projects · chat}};
\node[front] (cli) at (11.0,10.4) {\textbf{maestro-cli} (headless)\\[-1pt]{\scriptsize JSON output · NDJSON watch · CI exit codes}};

% ---------------- core ----------------
\node[core] (rules)  at (1.85,7.9) {\textbf{Rule engine}\\[-1pt]{\scriptsize roles $\rightarrow$ providers\\quota-aware fallbacks\\dry-run}};
\node[core] (orch)   at (5.75,7.9) {\textbf{Orchestrator}\\[-1pt]{\scriptsize write queue\\DAG scheduler\\git checkpoints}};
\node[core] (ledger) at (9.65,7.9) {\textbf{Ledger \& quota}\\[-1pt]{\scriptsize tokens · cost\\burn alerts}};
\node[core] (proj)   at (13.55,7.9) {\textbf{Projects}\\[-1pt]{\scriptsize interrogator\\PRD · AGENTS.md\\skills · task DAG}};

% ---------------- execution ----------------
\node[exec] (runtime) at (2.6,5.0) {\textbf{Agent runtime}\\[-1pt]{\scriptsize tool loop · sandboxed tools\\retry + migration}};
\node[exec] (adapters) at (7.7,5.0) {\textbf{CLI adapters}\\[-1pt]{\scriptsize PTY (ConPTY) · overlay\\harvest $\rightarrow$ tickets}};
\node[exec] (clients) at (12.8,5.0) {\textbf{Provider clients}\\[-1pt]{\scriptsize OpenAI · Anthropic · Ollama\\OAuth refresh · SSH tunnel}};

% ---------------- providers ----------------
\node[prov] (api)    at (2.6,2.5) {API subscriptions\\[-1pt]{\scriptsize kimi · anthropic · OpenAI-compatible}};
\node[prov] (ollama) at (7.7,2.5) {Ollama\\[-1pt]{\scriptsize local · cloud · remote over SSH}};
\node[prov] (agents) at (12.8,2.5) {CLI agents\\[-1pt]{\scriptsize claude · codex · gemini}};

% ---------------- the single-writer path ----------------
\node[files] (files) at (5.75,0.35) {\textbf{Workspace files}\\[-1pt]{\scriptsize atomic writes · git checkpoints}};

% ---------------- stores ----------------
\node[store] (cfg)   at (16.9,9.2) {config.toml\\rules.yaml};
\node[store] (keys)  at (16.9,7.9) {OS keychain\\age vault};
\node[store] (db)    at (16.9,6.6) {SQLite\\ledger · journal};

% ---------------- lanes ----------------
\begin{scope}[on background layer]
  \node[lane, fit=(tui)(cli)] (lane1) {};
  \node[lane, fit=(rules)(proj)] (lane2) {};
  \node[lane, fit=(runtime)(clients)] (lane3) {};
  \node[lane, fit=(api)(agents)] (lane4) {};
\end{scope}
\node[lanelbl] at ([xshift=2pt]lane1.north west) {front-ends};
\node[lanelbl] at ([xshift=2pt]lane2.north west) {core services (maestro-core)};

% ---------------- flows ----------------
